import { readFileSync } from 'node:fs';
import { readCfb, buildMetadata } from './src/index.ts';

const meta = buildMetadata(readCfb(readFileSync(process.argv[2])));
const word = (process.argv[3] ?? '').toLowerCase();
const one = (s) => String(s ?? '').replace(/\s+/g, ' ').trim();
for (const report of meta.reports ?? []) {
  const d = report.definition;
  if (!d) continue;
  console.log(`=== ${report.name ?? report.path ?? 'report'}`);
  console.log('GROUPS', JSON.stringify(d.groups), 'NAME FORMULAS', JSON.stringify(d.groupNameFormulas ?? []));
  for (const f of d.formulas) {
    const all = `${f.name} ${f.text}`.toLowerCase();
    if (/group/.test(all) || (word && all.includes(word))) console.log(`F [${f.kind}] ${f.name} = ${one(f.text).slice(0, 220)}`);
  }
  for (const area of d.layout) {
    if (!/group/i.test(area.name)) continue;
    for (const s of area.sections) {
      const cond = Object.entries(s.conditions ?? {}).map(([k, c]) => `${k}:${c.name}`).join(' ');
      console.log(`S ${area.name}/${s.name}${s.suppressed ? ' SUPPRESSED' : ''} ${cond}`);
      for (const o of s.objects) {
        const fields = [...(o.embeddedFields ?? []), ...(o.runs ?? []).filter((r) => 'field' in r).map((r) => r.field)];
        const oc = Object.entries(o.conditions ?? {}).map(([k, c]) => `${k}:${c.name}`).join(' ');
        console.log(`  O ${o.kind} ${o.name} x=${o.position?.x} ${o.field ?? ''} ${one(o.text).slice(0, 50)} ${fields.join(',')} ${oc}`);
      }
    }
  }
}
